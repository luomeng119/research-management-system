## 4. JS 逻辑：`static/js/equipment_index.js`

```js
// ========== 实时搜索 ==========
(function() {
    'use strict';
    
    const keywordInput = document.getElementById('keywordSearch');
    const tableWrap = document.getElementById('equipmentTableWrap');
    const loadingMask = document.getElementById('tableLoadingMask');
    const loadingIcon = document.getElementById('searchLoading');
    const successIcon = document.getElementById('searchSuccess');
    const errorBar = document.getElementById('searchError');
    const errorMsg = document.getElementById('searchErrorMsg');
    const retryBtn = document.getElementById('searchRetryBtn');
    const errorClose = document.getElementById('searchErrorClose');
    
    let debounceTimer = null;
    let lastFetchController = null;
    let pendingKeyword = null;
    
    function getQueryString() {
        const form = keywordInput.closest('form');
        const params = new URLSearchParams(new FormData(form));
        params.set('fragment', '1');
        // 移除空值
        for (const [k, v] of [...params.entries()]) {
            if (!v) params.delete(k);
        }
        return params.toString();
    }
    
    function doFetch() {
        const qs = getQueryString();
        if (lastFetchController) lastFetchController.abort();
        const controller = new AbortController();
        lastFetchController = controller;
        
        loadingIcon.style.display = 'inline';
        successIcon.style.display = 'none';
        errorBar.style.display = 'none';
        
        fetch('/equipment?' + qs, { signal: controller.signal })
            .then(r => {
                if (!r.ok) throw new Error('HTTP ' + r.status);
                return r.text();
            })
            .then(html => {
                if (controller.signal.aborted) return;
                // 解析 HTML 提取 #equipmentTableWrap
                const parser = new DOMParser();
                const doc = parser.parseFromString(html, 'text/html');
                const newWrap = doc.getElementById('equipmentTableWrap');
                if (newWrap) {
                    tableWrap.innerHTML = newWrap.innerHTML;
                }
                // URL 同步
                const url = new URL(window.location);
                const params = new URLSearchParams(qs);
                for (const [k, v] of params.entries()) url.searchParams.set(k, v);
                url.searchParams.delete('fragment');
                history.replaceState(null, '', url);
                // 反馈
                loadingIcon.style.display = 'none';
                successIcon.style.display = 'inline';
                setTimeout(() => { successIcon.style.display = 'none'; }, 1500);
            })
            .catch(err => {
                if (err.name === 'AbortError') return;
                loadingIcon.style.display = 'none';
                errorMsg.textContent = '⚠️ 检索失败：' + err.message + '。点击重试';
                errorBar.style.display = 'block';
                // 5s 自动消失
                setTimeout(() => { errorBar.style.display = 'none'; }, 5000);
            });
    }
    
    function onInput() {
        clearTimeout(debounceTimer);
        debounceTimer = setTimeout(doFetch, 300);
    }
    
    keywordInput.addEventListener('input', onInput);
    
    // Enter 键立即提交（绕过 debounce）
    keywordInput.addEventListener('keydown', function(e) {
        if (e.key === 'Enter') {
            e.preventDefault();
            clearTimeout(debounceTimer);
            doFetch();
        }
    });
    
    // 重置按钮触发检索
    const resetBtn = document.querySelector('a[href*="/equipment"]:not([href*="?"])');
    if (resetBtn) {
        resetBtn.addEventListener('click', function() {
            setTimeout(() => {
                keywordInput.value = '';
                doFetch();
            }, 100);
        });
    }
    
    // 重试
    retryBtn.addEventListener('click', doFetch);
    // 关闭错误条
    errorClose.addEventListener('click', () => { errorBar.style.display = 'none'; });
})();

// ========== 自定义 tooltip ==========
(function() {
    'use strict';
    
    const tooltip = document.getElementById('rowTooltip');
    const tbody = document.querySelector('#equipmentTableWrap tbody');
    if (!tbody || !tooltip) return;
    
    let showTimer = null;
    let currentRow = null;
    
    function show(row, e) {
        clearTimeout(showTimer);
        showTimer = setTimeout(() => {
            const html = row.getAttribute('data-tooltip-html');
            if (!html) return;
            // 按 \n 切分构造结构化内容
            const lines = html.split('\n').filter(l => l.trim());
            const content = lines.map(line => {
                const idx = line.indexOf(':');
                if (idx === -1) return `<div>${escapeHtml(line)}</div>`;
                const label = line.slice(0, idx).trim();
                const value = line.slice(idx + 1).trim();
                return `<div class="tt-row"><span class="tt-label">${escapeHtml(label)}:</span> <span class="tt-value">${escapeHtml(value)}</span></div>`;
            }).join('');
            tooltip.innerHTML = content;
            // 定位
            positionTooltip(e);
            tooltip.style.display = 'block';
            currentRow = row;
        }, 200);  // 200ms 延迟
    }
    
    function hide() {
        clearTimeout(showTimer);
        tooltip.style.display = 'none';
        currentRow = null;
    }
    
    function positionTooltip(e) {
        const offset = 15;
        let x = e.clientX + offset;
        let y = e.clientY + offset;
        const rect = tooltip.getBoundingClientRect();
        if (x + rect.width > window.innerWidth) {
            x = e.clientX - rect.width - offset;
        }
        if (y + rect.height > window.innerHeight) {
            y = e.clientY - rect.height - offset;
        }
        tooltip.style.left = x + 'px';
        tooltip.style.top = y + 'px';
    }
    
    function escapeHtml(s) {
        return s.replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
    }
    
    // 事件委托
    tbody.addEventListener('mouseover', function(e) {
        const tr = e.target.closest('tr');
        if (!tr || tr === currentRow) return;
        if (currentRow) hide();
        currentRow = tr;
        show(tr, e);
    });
    
    tbody.addEventListener('mousemove', function(e) {
        if (currentRow) positionTooltip(e);
    });
    
    tbody.addEventListener('mouseout', function(e) {
        const tr = e.target.closest('tr');
        if (!tr) return;
        // 检查鼠标是否真的离开了 tr
        const related = e.relatedTarget;
        if (!related || !tr.contains(related)) {
            hide();
        }
    });
    
    // 滚动/resize 隐藏
    window.addEventListener('scroll', hide, true);
    window.addEventListener('resize', hide);
})();
```

**关键设计**：
- **AbortController**：新请求 abort 旧请求，避免乱序响应覆盖
- **mouseover 委托**：AJAX 替换 innerHTML 后旧行的事件自然失效（因为旧 DOM 节点已销毁），无需重新绑定
- **mousemove 跟随**：保持 tooltip 跟随鼠标，体验更顺
- **mouseout 检测 relatedTarget**：避免移入子元素误触发 hide
- **事件委托**：`tbody` 一次绑定，AJAX 替换 tbody 后**新 tbody 不再触发 mouseover**！

⚠️ **关键问题**：AJAX 替换 `#equipmentTableWrap.innerHTML` 后，新的 tbody 没有 mouseover 监听。**修复**：把 tooltip 事件委托绑到 `#equipmentTableWrap`（容器）或 document（更稳）。

让我改：绑定到 `tableWrap`：

```js
tableWrap.addEventListener('mouseover', function(e) { ... });
```

这样 AJAX 替换 innerHTML 后，`tableWrap` DOM 节点不变，事件仍生效。
