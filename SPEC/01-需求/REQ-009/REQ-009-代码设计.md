```python
@dataclass

class DeviceMatch:
```

### 四级匹配实现

```python
def _match_by_name_or_model(text: str, table: str, id_col: str) -> MatchResult:
    # 精确匹配
    rows = db.execute(f"SELECT * FROM {table} WHERE name = ? OR model = ?", [text, text])
    if rows: return MatchResult(..., status='exact', ...)

    # 去噪声匹配（去除 "型号/" "设备" "V1" 等）
    cleaned = re.sub(r'[型号设备V\d.]+', '', text)
    rows = db.execute(f"SELECT * FROM {table} WHERE name LIKE ? OR model LIKE ?", [f'%{cleaned}%', f'%{cleaned}%'])
    if rows: return MatchResult(..., status='fuzzy', ...)

    # 相似度匹配（编辑距离 < 3 且 长度 > 5）
    all_rows = db.execute(f"SELECT * FROM {table}")
    candidates = [r for r in all_rows if levenshtein(text, r['name']) < 3 or text in r['name']]
    if candidates: return MatchResult(..., status='fuzzy', ...)

    # 未匹配
    return MatchResult(..., status='unmatched', ...)
```

### 多设备名称拆分

```python
def split_device_names(text: str) -> list[str]:
    if not text:
        return []
    # 按顿号、逗号、空格拆分
    parts = re.split(r'[、，, ]+', str(text))
    return [p.strip() for p in parts if p.strip()]
```

### 步骤1 POST 处理流程

```python
def import_preview():
    file = request.files['file']
    column_map = json.loads(request.form['column_mapping_json'])

    # 读取 Excel
    wb = openpyxl.load_workbook(BytesIO(file.read()), data_only=True)
    ws = wb.active

    # 识别关联字段列索引
    related_col = column_map.get('related_equipment')  # 或 'related_host_devices'

    processed_rows = []
    for row_idx, row in enumerate(ws.iter_rows(min_row=2, values_only=True)):
        if not row[0]: continue  # 跳过空行

        # 提取主字段
        record = extract_fields(row, column_map)

        # 关联设备匹配
        related_text = row[related_col] if related_col is not None else ''
        related_result = match_equipment(related_text) if target == 'equipment' else match_host_device(related_text)

        # 重复检测
        duplicate_status = check_duplicate(record)

        processed_rows.append({
            'row_idx': row_idx,
            **record,
            'related_equipment_result': related_result,
            'duplicate_status': duplicate_status
        })

    # 统计
    stats = compute_statistics(processed_rows)

    # 存入 session
    session['import_preview_data'] = {
        'filename': file.filename,
        'column_mapping': column_map,
        'processed': processed_rows,
        'statistics': stats
    }

    return jsonify({'success': True, 'redirect': url_for('host_devices.import_preview_page')})
```

```python
def import_commit():
    data = session.get('import_preview_data', {})
    corrections = request.form  # 用户在步骤2的修正数据

    imported = {'new': 0, 'overwrite': 0, 'skip': 0}
    relations_created = 0

    for row in data['processed']:
        row_idx = row['row_idx']

        # 读取用户修正后的值
        name = corrections.get(f'name_{row_idx}', row['name'])
        category = corrections.get(f'category_{row_idx}', row['category'])

        # 处理新增/覆盖/跳过

        if dup_action == 'skip':
            imported['skip'] += 1
            continue

        # 写入 host_devices 或 equipment
        if target == 'host_devices':
            if dup_action == 'overwrite':
                host_model.update(row['host_id'], name=name, category=category, ...)
            else:
                new_id = host_model.add(name=name, category=category, ...)
            imported['overwrite' if dup_action == 'overwrite' else 'new'] += 1
        else:
            ...

        # 处理关联关系
        for dev_match in row['related_equipment_result']['devices']:
            selected_id = corrections.get(f'select_{row_idx}_{loop.index}')
            if selected_id == '_skip_':
                continue
            # 写入 device_host_relations
            rel_model.add(device_id=selected_id, host_id=..., quantity=1)
            relations_created += 1

    return redirect(url_for('.import_done',

        skip=imported['skip'],
```
