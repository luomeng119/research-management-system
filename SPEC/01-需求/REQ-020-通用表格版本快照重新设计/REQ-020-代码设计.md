#### 新增 `save_snapshot`

```python
def save_snapshot(self, table_id, source_version_id, label, note, creator):
    """
    把 source 复制成 snapshot (is_locked=1) + 创建新 current (is_locked=0) + 更新 current_version_id
    返回 (snapshot_version_id, new_current_version_id)
    """
    with get_db() as conn:
        c = conn.cursor()
        # 1. 复制 source 为 snapshot
        snapshot_id = self.create_version(
            table_id=table_id, method='snapshot', source_version_id=source_version_id,
            creator=creator, label=label, note=note
        )
        # 2. 从 source 复制为新 current (method='import' 语义复用：不锁，列定义+数据复制)
        new_current_id = self.create_version(
            table_id=table_id, method='import', source_version_id=source_version_id,
            creator=creator, label=None, note='save_snapshot_new_working'
        )
        # 3. 更新 current_version_id
        c.execute('UPDATE generic_tables SET current_version_id = ? WHERE table_id = ?',
                  (new_current_id, table_id))
        conn.commit()
        return snapshot_id, new_current_id
```

```python
def rollback_to(self, table_id, source_version_id, creator):
    """从 source 派生新 current (method='rollback', is_locked=0) + 更新 current_version_id"""
    with get_db() as conn:
        c = conn.cursor()
        # 创建新 current
        new_current_id = self.create_version(
            table_id=table_id, method='import', source_version_id=source_version_id,
            creator=creator, label=None, note='rollback_to_v{N}'
        )
        # 更新 current_version_id
        c.execute('UPDATE generic_tables SET current_version_id = ? WHERE table_id = ?',
                  (new_current_id, table_id))
        conn.commit()
        return new_current_id
```

async function importExcelNew(input) {
    if (!input.files[0]) return;
    const file = input.files[0];
    const formData = new FormData();
    formData.append('file', file);
    formData.append('mode', 'replace');
    
    try {
        // REQ-020: 直接导入到 current_version（不再先建 snapshot）
        const res = await fetch(`/api/generic-tables/versions/${VERSION_ID}/rows/import`, {
            method: 'POST',
            body: formData
        });
        const data = await res.json();
        
        if (data.error) {
            showAlert('danger', data.error);
            return;
        }
        
        showAlert('success', `成功导入 ${data.imported} 条数据`);
        // 刷新页面让前端重新加载 current 数据
        setTimeout(() => location.reload(), 800);
    } catch (e) {
        showAlert('danger', '导入失败：' + e.message);
    } finally {
        input.value = '';  // 清空 file input，允许重选同一文件
    }
}
```

function openSaveSnapshotModal() {
    const label = prompt('请输入快照名称（必填，最长 50 字）：');
    if (!label || !label.trim()) return;
    if (label.length > 50) { alert('名称过长'); return; }
    
    const note = prompt('请输入快照说明（选填，最长 500 字）：') || '';
    if (note.length > 500) { alert('说明过长'); return; }
    
    saveSnapshot(label.trim(), note.trim());
}

async function saveSnapshot(label, note) {

        if (data.error) { showAlert('danger', data.error); return; }
        showAlert('success', '快照保存成功');
        setTimeout(() => location.reload(), 800);

        showAlert('danger', '保存失败：' + e.message);
    }
}

        location.href = `/generic-tables/${TABLE_ID}/`;
        return;
    }
    location.href = `/generic-tables/${TABLE_ID}/version/${vid}/`;
}
```

## 6. 数据迁移

### 6.1 启动时自动迁移（`init_db` 内）

见 §2.2。**幂等**：跑一次后第二次跑不会重复修改。

### 6.2 手动迁移脚本（备份用）

```python
# scripts/migrate_req020.py
"""REQ-020 数据迁移脚本（手动跑一次）"""
import sqlite3
import sys

DB_PATH = sys.argv[1] if len(sys.argv) > 1 else 'data/research.db'

conn = sqlite3.connect(DB_PATH)
c = conn.cursor()

# 1. 加字段
try:
    c.execute('ALTER TABLE generic_tables ADD COLUMN current_version_id TEXT')
except Exception:
    pass

# 2. 回填 current
c.execute('''
    UPDATE generic_tables
    SET current_version_id = (
        SELECT version_id FROM generic_table_versions
        WHERE table_id = generic_tables.table_id AND is_locked = 0
        ORDER BY version_number DESC LIMIT 1
    )
    WHERE current_version_id IS NULL OR current_version_id = ''
''')

# 3. 兜底：所有版本都锁定 → 取最大版本号
c.execute('''
    UPDATE generic_tables
    SET current_version_id = (
        SELECT version_id FROM generic_table_versions
        WHERE table_id = generic_tables.table_id
        ORDER BY version_number DESC LIMIT 1
    )
    WHERE current_version_id IS NULL OR current_version_id = ''
''')

# 4. 兜底：没有版本
c.execute('''
    INSERT INTO generic_table_versions
    (version_id, table_id, version_number, version_label, create_method, row_count, creator, created_at, is_locked)
    SELECT 'GTV-' || table_id || '-migrate', table_id, 1, 'v1', 'create', 0, 'migrate', datetime('now'), 0
    FROM generic_tables
    WHERE NOT EXISTS (SELECT 1 FROM generic_table_versions WHERE generic_table_versions.table_id = generic_tables.table_id)
''')

c.execute('''
    UPDATE generic_tables
    SET current_version_id = 'GTV-' || table_id || '-migrate'
    WHERE current_version_id IS NULL OR current_version_id = ''
''')

conn.commit()

# 验证
c.execute('SELECT COUNT(*) FROM generic_tables WHERE current_version_id IS NULL OR current_version_id = ""')
unmatched = c.fetchone()[0]
print(f'迁移完成，未匹配 current 的表格数：{unmatched}（应为 0）')
conn.close()
```

## 7. 关键风险点

### 7.1 save_snapshot 性能

- 每次保存快照 = 复制列定义 + 复制所有行
- 1000 行数据复制 = 几十毫秒
- 1 万行数据复制 = 几百毫秒（用户能感知）
- **应对**：save_snapshot 在前端用 loading 遮罩，禁止连点

### 7.2 current_version_id 引用一致性

- 任何写操作都要保证 current_version_id 指向有效版本
- **应对**：用数据库外键约束 + 启动时校验

```sql
-- 暂不加外键（SQLite 外键约束需 PRAGMA），代码层校验
```

### 7.3 历史数据 import 标记

- 老数据中可能有 `create_method='import'` 的版本（REQ-019 之前可能是 is_locked=1）
- 迁移脚本会保留老状态
- 新设计 `create_method='import'` 含义变了（不是用户导入的快照，是内部 working 副本）
- **应对**：迁移时给老 `import` 版本改 create_method='legacy_import'（保留历史语义）

### 7.4 URL 兼容

- 老 URL `/generic-tables/<id>/?version=<vid>` 走 GET 时默认 `is_current=True`（按 current 处理），不识别老参数
- **应对**：保留老参数支持，传入 `version_id` 时切到只读模式

```python
# GET /generic-tables/<table_id>/?version=<vid> 兼容
table_id = ...
requested_vid = request.args.get('version')
if requested_vid:
    return version_readonly_page(table_id, requested_vid)
else:
    return detail_page(table_id)
```
